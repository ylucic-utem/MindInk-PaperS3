current processing status\nWITH audio_status AS (\n  SELECT \n    id,\n    file_name,\n    status,\n    summary_id,\n    recording_date\n  FROM audio_records\n  WHERE recording_date > NOW() - INTERVAL '1 hour'\n  ORDER BY recording_date DESC\n)\nSELECT \n  a.file_name,\n  a.status as audio_status,\n  s.status as summary_status,\n  i.image_file_name,\n  EXTRACT(EPOCH FROM (NOW() -
