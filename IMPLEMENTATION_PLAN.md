{ audioFileId } = await req.json();\n  \n  // Create Supabase client with service role\n  const supabase = createClient(\n    Deno.env.get('SUPABASE_URL')!,\n    Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!\n  );\n  \n  try {\n    // 1. Get audio file metadata\n    const { data: audioFile } = await supabase\n      .from('audio_files')\n      .select('*')\n      .eq('id', audioFileId)\n      .single();\n    \n    // 2. Download audio file from storage\n    const { data: audioData } = await supabase\n      .storage\n      .from('audios')\n      .download(audioFile.storage_path);\n    \n    // 3. Send to ElevenLabs for transcription\n    const transcriptionResponse = await fetch(\n      'https://api.elevenlabs.io/v1/speech-to-text',\n      {\n        method: 'POST',\n        headers: {\n          'xi-api-key': ELEVEN_LABS_API_KEY!,\n        },\n        body: audioData,\n      }\n    );\n    \n    const { text: transcription } = await transcriptionResponse.json();\n    \n    // 4. Save summary to database\n    const { data: summary } = await supabase\n      .from('summaries')\n      .insert({\n        audio_file_id: audioFileId,\n        summary_text: summaryText,\n        created_by: audioFile.uploaded_by,\n        word_count: summaryText.split(' ').length\n      })\n      .select()\n      .single();\n    \n    // 6.
